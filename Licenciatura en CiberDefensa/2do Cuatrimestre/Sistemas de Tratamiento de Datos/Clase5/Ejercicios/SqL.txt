-- RESOLUCIÓN CREACIÓN TABLAS (Puntos 1 a 4) 

-- Punto 1: 
CREATE TABLE Artista 
( artistaID INT PRIMARY KEY, 
 nombre VARCHAR(50) NOT NULL UNIQUE, -- Se agrega NOT NULL y UNIQUE 
 pais VARCHAR(50), 
 fechaNacimiento DATE 
); 

-- Punto 2: 
CREATE TABLE Album 
( albumID INT PRIMARY KEY, 
 titulo VARCHAR(255) NOT NULL, -- Se agrega NOT NULL 
 artistaID INT REFERENCES Artista(artistaID), 
 anioLanzamiento INT 
); 

-- Punto 3: 
CREATE TABLE Canciones 
( cancionID INT PRIMARY KEY, 
 titulo VARCHAR(255), 
 albumID INT, 
 genero VARCHAR(50) DEFAULT 'Pop', -- Se agrega DEFAULT 'Pop'
 duracionSegundos INT, 
 FOREIGN KEY (albumID) REFERENCES Album(albumID) 
); 

-- Punto 4: 
CREATE TABLE Ventas 
( ventaID INT, 
 cancionID INT, 
 fechaVenta DATE, 
 cantidad INT DEFAULT 1, -- Se agrega DEFAULT 1 
 precioUnitario DECIMAL(10, 2), 
 PRIMARY KEY (ventaID), 
 CONSTRAINT FK_Canciones FOREIGN KEY (cancionID) REFERENCES 
Canciones(cancionID) 
);

-- Punto 20:

SELECT pais, COUNT(*) AS total 
 FROM Artista 
 GROUP BY pais 
HAVING COUNT(*) > 1; 

-- Punto 22:

-- Punto 23: 
SELECT cancionID, SUM(cantidad * preciounitario) as montoTotal 
 FROM Ventas

-- Punto 24: 
SELECT Album.titulo 
 FROM Album INNER JOIN Artista on Album.artistaID = Artista.artistaID 
 WHERE Artista.pais = 'Estados Unidos' 
 ORDER BY Album.titulo ASC; -- Por defecto se ordena ascendente (incluso sin ASC) 

-- Punto 25: 
SELECT Canciones.titulo, Canciones.genero 
 FROM Canciones 
 INNER JOIN Album ON Canciones.albumID = Album.albumID 
 INNER JOIN Artista on Album.artistaID = Artista.artistaID 
 WHERE Artista.nombre = 'The Beatles';

 GROUP BY Album.titulo
